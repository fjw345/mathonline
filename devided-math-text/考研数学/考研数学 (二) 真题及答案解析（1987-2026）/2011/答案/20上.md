$$
\begin{array}{l} V = V _ {1} + V _ {2} = \pi \int_ {\frac {1}{2}} ^ {2} (2 y - y ^ {2}) d y + \pi \int_ {- 1} ^ {\frac {1}{2}} (1 - y ^ {2}) d y \\ = \pi \left(y ^ {2} - \frac {y ^ {3}}{3}\right) \Bigg | _ {\frac {1}{2}} ^ {2} + \pi \left(y - \frac {y ^ {3}}{3}\right) \Bigg | _ {- 1} ^ {\frac {1}{2}} = \pi \left(5 + \frac {1}{4} - 3\right) = \frac {9}{4} \pi . \end{array}
$$
