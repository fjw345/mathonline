原式

$$
\begin{array}{l} = \int_ {1} ^ {+ \infty} \frac {a}{x (2 x + a)} \mathrm{d} x = \ln x - \ln (2 x + a) \Big | _ {1} ^ {+ \infty} = \int_ {1} ^ {+ \infty} \frac {a}{x (2 x + a)} \mathrm{d} x = \ln x - \ln (2 x + a) \Big | _ {1} ^ {+ \infty} \\ = \lim _ {x \to + \infty} \ln \frac {x}{2 x + a} - \ln \frac {1}{2 + a} = \lim _ {x \to + \infty} \frac {x}{x + a} + \ln (2 + a) \end{array}
$$

$$
\begin{array}{l} \therefore \ln \frac {1}{2} + \ln (2 + a) = \ln 2 \\ \ln (2 + a) = 2 \ln 2 \Rightarrow a = 2 \end{array}
$$
