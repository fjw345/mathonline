$$
\begin{array}{r l} & {\text {(2) 体积:} f (x) = \frac {x}{\sqrt {x ^ {2} + 1}} \Rightarrow x = \frac {y}{\sqrt {1 - y ^ {2}}}} \\ & {\qquad V = 2 \pi \int_ {\frac {1}{2}} ^ {\frac {\sqrt {3}}{2}} y g (y) \mathrm{d} y} \\ & {\qquad = 2 \pi \int_ {\frac {1}{2}} ^ {\frac {\sqrt {3}}{2}} \frac {y ^ {2}}{\sqrt {1 - y ^ {2}}} \mathrm{d} y} \\ & {\qquad \stackrel {y = \sin t} {=} 2 \pi \int_ {\frac {\pi}{6}} ^ {\frac {\pi}{3}} \frac {\sin^ {2} t}{\cos t} \cdot \cos t \mathrm{d} t} \\ & {\qquad = 2 \pi \int_ {\frac {\pi}{6}} ^ {\frac {\pi}{3}} \sin^ {2} t \mathrm{d} t = 2 \pi \int_ {\frac {\pi}{6}} ^ {\frac {\pi}{3}} \frac {1 - \cos 2 t}{2} \mathrm{d} t = \pi \Biggl [ \left(\frac {\pi}{3} - \frac {\pi}{6}\right) - \frac {1}{2} \sin 2 t \Biggr | _ {\frac {\pi}{6}} ^ {\frac {\pi}{3}} \Biggr ]} \\ & {\qquad = \pi \biggl (\frac {\pi}{6} - \frac {1}{2} \times 0 \biggr) = \frac {\pi^ {2}}{6}} \end{array}
