$$
\begin{array}{l} \lim _ {x \to \pi -} \frac {F (x) - F (\pi)}{x - \pi} = \lim _ {x \to \pi +} \frac {- 1 - \cos x}{x - \pi} = 0, \quad \lim _ {x \to \pi +} \frac {F (x) - F (\pi)}{x - \pi} = \lim _ {x \to \pi +} \frac {2 (x - \pi)}{x - \pi} = 2, \\ \text {所以} F (x) \text {在} x = \pi \text {处不可导。} \end{array}
