$$
\begin{array}{l} \lim _ {x \to 0 ^ {+}} \frac {\ln^ {\alpha} (1 + 2 x)}{x} = \lim _ {x \to 0 ^ {+}} \frac {(2 x) ^ {\alpha}}{x} = 2 ^ {\alpha} \lim _ {x \to 0 ^ {+}} x ^ {\alpha - 1} = 0 \\ \lim _ {x \to 0 ^ {+}} \frac {(1 - \cos x) ^ {\frac {1}{2}}}{x} = \lim _ {x \to 0 ^ {+}} \frac {(\frac {1}{2} x ^ {2}) ^ {\frac {1}{\alpha}}}{x} = (\frac {1}{2}) ^ {\frac {1}{\alpha}} \lim _ {x \to 0 ^ {+}} x ^ {\frac {2}{a} - 1} = 0 \\ \therefore \frac {2}{\alpha} - 1 > 0 \therefore \alpha < 2 \end{array}
