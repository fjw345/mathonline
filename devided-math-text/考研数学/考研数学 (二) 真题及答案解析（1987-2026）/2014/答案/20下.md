$$
f (\mathrm{x}) = \frac {X}{1 + X}, f _ {1} (\mathrm{x}) = \mathrm{f(x)}
$$

$$
f _ {2} (\mathrm{x}) = f (\mathrm{f} _ {1} (\mathrm{x})) = \frac {\frac {X}{1 + X}}{1 + \frac {X}{1 + X}} = \frac {X}{1 + 2 x}
$$

$$
\left| f _ {3} (\mathrm{x}) = f (f _ {2} (\mathrm{x})) = \frac {\frac {X}{1 + 2 x}}{1 + \frac {x}{1 + 2 x}} = \frac {X}{1 + 3 x} \right|
$$

用归纳法知: $f_{\mathrm{n}}(\mathrm{x}) = \frac{X}{1 + nx}, x \in [0,1]$

$$
\begin{array}{l} \mathrm{S} _ {n} = \int_ {0} ^ {1} \frac {x}{1 + n x} d x = \frac {1}{n} \int_ {0} ^ {1} \frac {n x + 1 - 1}{1 + n x} d x \\ \quad = \frac {1}{n} \int_ {0} ^ {1} (1 - \frac {1}{1 + n x}) d x \\ \quad = \frac {1}{n} - \frac {1}{n ^ {2}} \ln (1 + n) \\ \lim _ {n \to \infty} n \mathrm{S} _ {n} = \lim _ {n \to \infty} n [ \frac {1}{n} - \frac {1}{n ^ {2}} \ln (1 + n) ] = 1 - \lim _ {n \to \infty} \frac {\ln (1 + n)}{n} \\ \quad = 1 \end{array}
$$
