$$
\left| \begin{array}{c c c c} 1 & a & 0 & 0 \\ 0 & 1 & a & 0 \\ 0 & 0 & 1 & a \\ a & 0 & 0 & 1 \end{array} \right| = 1 \times \left| \begin{array}{c c c} 1 & a & 0 \\ 0 & 1 & a \\ 0 & 0 & 1 \end{array} \right| + a \times (- 1) ^ {4 + 1} \left| \begin{array}{c c c} a & 0 & 0 \\ 1 & a & 0 \\ 0 & 1 & a \end{array} \right| = 1 - a ^ {4}
$$

$$

$$
\rightarrow \left(\begin{array}{c c c c c}1&a&0&0&1\\0&1&a&0&- 1\\0&0&1&a&0\\0&0&0&1 - a ^ {4}&- a - a ^ {2}\end{array}\right)
