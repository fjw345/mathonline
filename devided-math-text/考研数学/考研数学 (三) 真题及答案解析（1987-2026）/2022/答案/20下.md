$$
(\sum_ {n = 0} ^ {\infty} \frac {(- 1) ^ {n}}{2 n + 1} x ^ {2 n + 1}) ^ {\prime} = \sum_ {n = 0} ^ {\infty} (- 1) ^ {n} x ^ {2 n} = \frac {1}{1 + x ^ {2}}.
$$

$$
\sum_ {n = 0} ^ {\infty} \frac {(- 1) ^ {n}}{2 n + 1} x ^ {2 n + 1} = \int_ {0} ^ {x} \frac {1}{1 + t ^ {2}} d t = \arctan x.
$$

则 $S_{1}(x) = \frac{1}{x}\arctan x, x \neq 0$

$$
S _ {2} (x) = \sum_ {n = 0} ^ {\infty} \frac {x ^ {2 n}}{4 ^ {n} (2 n + 1)} = \frac {1}{x} \sum_ {n = 0} ^ {\infty} \frac {x ^ {2 n + 1}}{4 ^ {n} (2 n + 1)}, x \neq 0,
$$

$$
(\sum_ {n = 0} ^ {\infty} \frac {x ^ {2 n + 1}}{4 ^ {n} (2 n + 1)}) ^ {\prime} = \sum_ {n = 0} ^ {\infty} \frac {x ^ {2 n}}{4 ^ {n}} = \frac {1}{1 - \frac {x ^ {2}}{4}} = \frac {4}{4 - x ^ {2}},
$$

$$
\sum_ {n = 0} ^ {\infty} \frac {x ^ {2 n + 1}}{4 ^ {n} (2 n + 1)} = \int_ {0} ^ {x} \frac {4}{4 - t ^ {2}} d t = \ln \frac {2 + x}{2 - x}, S _ {2} (x) = \frac {1}{x} \ln \frac {2 + x}{2 - x}, x \neq 0.
$$

$$
S (0) = 2.
$$

综上 $S(x) = \left\{ \begin{array}{cccc} \frac{1}{x}\arctan x + \frac{1}{x}\ln \frac{2 + x}{2 - x}, & x \neq 0, & x \in [-1,1] \\ 2, & x = 0 \end{array} \right.$.
