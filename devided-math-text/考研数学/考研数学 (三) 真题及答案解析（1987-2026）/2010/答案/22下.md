$$
\begin{array}{l} \int_ {- \infty} ^ {+ \infty} \frac {1}{\sqrt {2 \pi} \sigma} \mathrm{e} ^ {- \frac {(t - \mu) ^ {2}}{2 \sigma^ {2}}} \mathrm{d} t = 1, \\ 1 = \int_ {- \infty} ^ {+ \infty} \int_ {- \infty} ^ {+ \infty} f (x, y) \mathrm{d} x \mathrm{d} y = A \int_ {- \infty} ^ {+ \infty} \int_ {- \infty} ^ {+ \infty} \mathrm{e} ^ {- 2 x ^ {2} + 2 x y - y ^ {2}} \mathrm{d} x \mathrm{d} y \\ = A \int_ {- \infty} ^ {+ \infty} \mathrm{e} ^ {- x ^ {2}} \mathrm{d} x \int_ {- \infty} ^ {+ \infty} \mathrm{e} ^ {- (\gamma - x) ^ {2}} \mathrm{d} y = A \int_ {- \infty} ^ {+ \infty} \mathrm{e} ^ {- x ^ {2}} \mathrm{d} x \int_ {- \infty} ^ {+ \infty} \frac {\sqrt {\pi}}{\sqrt {2 \pi} \sqrt {\frac {1}{2}}} \mathrm{e} ^ {- \frac {(\gamma - x) ^ {2}}{2 \cdot (\frac {1}{\sqrt {2}}) ^ {2}}} \mathrm{d} y \\ = A \sqrt {\pi} \int_ {- \infty} ^ {+ \infty} \mathrm{e} ^ {- x ^ {2}} \mathrm{d} x = A \sqrt {\pi} \int_ {- \infty} ^ {+ \infty} \frac {\sqrt {\pi}}{\sqrt {2 \pi} \sqrt {\frac {1}{2}}} \mathrm{e} ^ {- \frac {x ^ {2}}{2 (\frac {1}{\sqrt {2}}) ^ {2}}} \mathrm{d} x = A \pi , \end{array}
