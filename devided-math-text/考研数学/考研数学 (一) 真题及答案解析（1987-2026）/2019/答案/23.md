【解析】(I)

$$
\begin{array}{l} {\ln L = \sum_ {i = 1} ^ {n} (\ln \frac {1}{2} - \ln \sigma - \frac {| x _ {i} |}{\sigma})} \\ {\therefore \text {令} \frac {d \ln L}{d \sigma} = \sum_ {i = 1} ^ {n} (- \frac {1}{\sigma} + \frac {| x _ {i} |}{\sigma^ {2}}) = 0 \Rightarrow \hat {\sigma} = \frac {1}{n} \sum_ {i = 1} ^ {n} | X _ {i} |} \end{array}
$$

$$
E \hat {\sigma} = \frac {1}{n} \sum_ {i = 1} ^ {n} E \left| X _ {i} \right| = E \left| X \right| = \int_ {- \infty} ^ {+ \infty} \frac {\left| X \right|}{2 \sigma} e ^ {- \frac {\left| X \right|}{\sigma}} d x = \int_ {0} ^ {+ \infty} \frac {x}{\sigma} e ^ {- \frac {x}{\sigma}} d x = \sigma\tag{II}
$$

$$
\begin{array}{c} D \hat {\sigma} = \frac {1}{n ^ {2}} \sum_ {i = 1} ^ {n} D \left| X _ {i} \right| = \frac {1}{n} D \left| X \right| = \frac {1}{n} (E X ^ {2} - E ^ {2} \left| X \right|) = \frac {1}{n} (\int_ {- \infty} ^ {+ \infty} \frac {x ^ {2}}{2 \sigma} e ^ {- \frac {| x |}{\sigma}} d x - \sigma^ {2}) \\ = \frac {1}{n} (\int_ {0} ^ {+ \infty} \frac {x ^ {2}}{\sigma} e ^ {- \frac {x}{\sigma}} d x - \sigma^ {2}) = \frac {\sigma^ {2}}{n}. \end{array}
$$
