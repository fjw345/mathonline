$$
(\alpha_ {1}, \dots , \alpha_ {n}) \left( \begin{array}{c c c} b _ {1 1} & ... & b _ {1 n} \\ . & ... & . \\ b _ {n 1} & ... & b _ {n n} \end{array} \right) = (\gamma_ {1}, \dots , \gamma_ {n})
$$
