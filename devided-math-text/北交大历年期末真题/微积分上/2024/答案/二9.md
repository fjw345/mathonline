\begin{array}{r l} \text {解:} & \lim _ {x \to \pi} \frac {e ^ {\pi} - e ^ {x}}{\sin 3 x - \sin x} = \lim _ {x \to \pi} \frac {e ^ {x} (e ^ {\pi - x} - 1)}{\sin 3 x - \sin x} = e ^ {\pi} \lim _ {x \to \pi} \frac {\pi - x}{\sin 3 x - \sin x} \\ & = e ^ {\pi} \lim _ {x \to \pi} \frac {- 1}{3 \cos 3 x - \cos x} = \frac {- e ^ {\pi}}{2 \cos \pi} = \frac {e ^ {\pi}}{2}. \end{array}
