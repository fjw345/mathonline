$$
\begin{array}{l} \left| \lim _ {x \to \infty} e ^ {- x} \left(\mathbf {1} + \frac {\mathbf {1}}{x}\right) ^ {x ^ {2}} = \lim _ {x \to \infty} \left[ \left(1 + \frac {1}{x}\right) ^ {x} e ^ {- 1} \right] ^ {x} \right. \\ = \exp \left(\lim _ {x \to \infty} \left[ \ln \left(1 + \frac {1}{x}\right) ^ {x} - 1 \right] x\right) = \exp \left(\lim _ {x \to \infty} x \left[ x \ln \left(1 + \frac {1}{x}\right) - 1 \right]\right) \\ = \exp \left(\lim _ {x \to \infty} x \left[ x \left(\frac {1}{x} - \frac {1}{2 x ^ {2}} + o (\frac {1}{x ^ {2}})\right) - 1 \right]\right) = e ^ {- \frac {1}{2}}. \end{array}
$$
