$\therefore \sin \left(\pi \sqrt {1 + 4 n ^ {2}}\right) = \sin \left(\pi \sqrt {1 + 4 n ^ {2}} - 2 n \pi\right) = \sin \frac {\pi}{2 n \pi + \pi \sqrt {1 + 4 n ^ {2}}}\tag{2$分$}$

$$
\begin{array}{r l} \text {原式} & = \lim _ {n \to \infty} \left(1 + \sin \frac {\pi}{2 n \pi + \pi \sqrt {1 + 4 n ^ {2}}}\right) ^ {n} \\ & = \exp \left[ \lim _ {n \to \infty} n \ln \left(1 + \sin \frac {\pi}{2 n \pi + \pi \sqrt {1 + 4 n ^ {2}}}\right) \right] \\ & = \exp \left(\lim _ {n \to \infty} n \sin \frac {\pi}{2 n \pi + \pi \sqrt {1 + 4 n ^ {2}}}\right) \\ & = \exp \left(\lim _ {n \to \infty} \frac {\pi n}{2 n \pi + \pi \sqrt {1 + 4 n ^ {2}}}\right) = e ^ {\frac {1}{4}} \end{array}\tag{2分}
$$
